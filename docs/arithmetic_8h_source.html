47 template <typename RingLike>
48 inline void SparseVectorAddition(
49 SparseVector<RingLike>& u,
50 const SparseVector<RingLike>& v,
51 Ring* R,
52 ComputeResources<RingLike>& resources
53 ) {
54 // If one of the vectors is zero, just return.
55 if (v.nnz() == 0) return;
56 if (u.nnz() == 0) return;
57
58 // Start from the largest index of each vector.
59 size_t left = u.nnz();
60 size_t right = v.nnz();
61 size_t out = u.nnz() + v.nnz();
62
63 // Preallocate enough space to accommodate all entries in both vectors.
64 if (u.alloc() < out) u.reserve(out);
65 u.resize(out);
66
67 while (left > 0 && right > 0) {
68 // If we're pointing to the same index...
69 if (u(left-1) == v(right-1)) {
70 auto entry = resources.arithmetic->add[(int)u[left-1]][(int)v[right-1]];
71
72 // ... and we get a nonzero entry, then the index of this entry
73 // is the same as in the original vector, and the entry is the
74 // result of the arithmetic; we then decrement the `out` pointer...
75 if (entry != 0) {
76 u(out-1) = u(left-1);
77 u[out-1] = entry;
78 out--;
79 }
80
81 // ... and decrement both left and right pointers, since we
82 // found a result in both.
83 left--;
84 right--;
85 }
86
87 // If we *aren't* pointing at the same index, and the index on the
88 // left is smaller than the index on the right...
89 else if (u(left-1) < v(right-1)) {
90 // ... then we set the index of this entry in the output vector
91 // to the index of the one on the right (since this preserves
92 // the order), then set the entry...
93 u(out-1) = v(right-1);
94 u[out-1] = v[right-1];
95
96 // ... then decrement the right pointer and the out pointer,
97 // since they track together in this case.
98 right--;
99 out--;
100 }
101
102 // Otherwise, the index on the right is smaller than the one on
103 // the left, and we do the same thing as the "else if" from above
104 // but with the roles reversed.
105 else {
106 u(out-1) = u(left-1);
107 u[out-1] = u[left-1];
108 left--;
109 out--;
110 }
111 }
112
113 // Then, while the right pointer is still bigger than zero, write
114 // the remaining entries into the vector.
115 while (right > 0) {
116 u(out-1) = v(right-1);
117 u[out-1] = v[right-1];
118 right--;
119 out--;
120 }
121
122 // Zero out the remaining entries, if there are any.
123 // #pragma omp parallel for default(shared) schedule(guided)
124 for (size_t i=left; i < out; i++) u[i] = 0;
125
126 // Canonicalize.
127 u.canonicalize();
128 }
